import org.springframework.web.bind.annotation.*;

import java.util.Optional;

@RestController
@RequestMapping("/auth")
public class AuthController {

    @Autowired
    private ContractorRepository contractorRepository;

    private JwtUtil jwtUtil = new JwtUtil();

    // LOGIN API
    @PostMapping("/login")
    public String login(@RequestParam String phone) {

        Optional<Contractor> contractor =
                contractorRepository.findAll()
                        .stream()
                        .filter(c -> phone.equals(c.getPhone()))
                        .findFirst();

        if (contractor.isEmpty()) {
            return "Invalid phone number";
        }

        // Generate JWT token
        String token = jwtUtil.generateToken(contractor.get().getPhone());

        return token;
    }

	package com.labourlink.labourlinkai.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
public class JwtFilter extends OncePerRequestFilter {

    private JwtUtil jwtUtil = new JwtUtil();

    @Override
    protected void doFilterInternal(
            HttpServletRequest request,
            HttpServletResponse response,
            FilterChain filterChain
    ) throws ServletException, IOException {

        String path = request.getRequestURI();

        // Public endpoints (NO JWT required)
        if (path.startsWith("/auth")
                || path.startsWith("/contractors")
                || path.startsWith("/workers")) {

            filterChain.doFilter(request, response);
            return;
        }

        String authHeader = request.getHeader("Authorization");

        // Check Authorization header
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            response.getWriter().write("Missing or invalid token");
            return;
        }

        String token = authHeader.substring(7);

        // Validate JWT token
        if (!jwtUtil.validateToken(token)) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            response.getWriter().write("Invalid token");
            return;
        }

        // Token valid -> continue request
        filterChain.doFilter(request, response);
    }
}

	package com.labourlink.labourlinkai.security;

import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.SignatureAlgorithm;
import io.jsonwebtoken.security.Keys;

import java.security.Key;
import java.util.Date;

public class JwtUtil {

    // Secret key (VERY IMPORTANT)
    private static final String SECRET = "my_super_secret_key_for_labourlink_ai_project_123456";

    private static final long EXPIRATION_TIME = 1000 * 60 * 60 * 10; // 10 hours

    private Key getSigningKey() {
        return Keys.hmacShaKeyFor(SECRET.getBytes());
    }

    // Generate token
    public String generateToken(String username) {

        return Jwts.builder()
                .setSubject(username)
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + EXPIRATION_TIME))
                .signWith(getSigningKey(), SignatureAlgorithm.HS256)
                .compact();
    }

    // Extract username
    public String extractUsername(String token) {
        return Jwts.parserBuilder()
                .setSigningKey(getSigningKey())
                .build()
                .parseClaimsJws(token)
                .getBody()
                .getSubject();
    }

    // Validate token
    public boolean validateToken(String token) {
        try {
            extractUsername(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}

package com.labourlink.labourlinkai.service;

import com.labourlink.labourlinkai.entity.Labour;
import com.labourlink.labourlinkai.repository.LabourRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;
import java.util.stream.Collectors;

@Service
public class LabourServiceImpl implements LabourService {

    @Autowired
    private LabourRepository labourRepository;

    // CREATE
    @Override
    public Labour saveLabour(Labour labour) {
        return labourRepository.save(labour);
    }

    // READ ALL
    @Override
    public List<Labour> getAllLabours() {
        return labourRepository.findAll();
    }

    // UPDATE
    @Override
    public Labour updateLabour(Long id, Labour updatedLabour) {

        Optional<Labour> existing = labourRepository.findById(id);

        if (existing.isPresent()) {

            Labour labour = existing.get();

            labour.setName(updatedLabour.getName());
            labour.setPhone(updatedLabour.getPhone());
            labour.setSkill(updatedLabour.getSkill());
            labour.setLocation(updatedLabour.getLocation());
            labour.setAvailable(updatedLabour.isAvailable());
            labour.setRating(updatedLabour.getRating());

            return labourRepository.save(labour);
        }

        throw new RuntimeException("Labour not found with id: " + id);
    }

    // DELETE
    @Override
    public void deleteLabour(Long id) {
        labourRepository.deleteById(id);
    }

    // FILTER BY SKILL
    @Override
    public List<Labour> filterBySkill(String skill) {
        return labourRepository.findBySkillIgnoreCase(skill);
    }

    // FILTER BY LOCATION
    @Override
    public List<Labour> filterByLocation(String location) {
        return labourRepository.findByLocationIgnoreCase(location);
    }

    // SEARCH + RANKING ENGINE
    @Override
    public List<Labour> searchAndRankLabours(String skill, String location) {

        List<Labour> labours = labourRepository.findAll();

        // STEP 1: only available workers
        labours = labours.stream()
                .filter(Labour::isAvailable)
                .collect(Collectors.toList());

        // STEP 2: ranking
        labours.sort((a, b) ->
                calculateScore(b, skill, location) - calculateScore(a, skill, location)
        );

        return labours;
    }

    // SCORING ENGINE (SMART RANKING)
    private int calculateScore(Labour labour, String skill, String location) {

        int score = 0;

        // skill match
        if (labour.getSkill() != null &&
                labour.getSkill().equalsIgnoreCase(skill)) {
            score += 50;
        }

        // location match
        if (labour.getLocation() != null &&
                labour.getLocation().equalsIgnoreCase(location)) {
            score += 30;
        }

        // rating boost (capped)
        score += Math.min((int) (labour.getRating() * 5), 25);

        return score;
    }
}
